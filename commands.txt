git init — создаёт новый локальный репозиторий
git status — показывает состояние файлов
git add — добавляет файлы в индекс
git commit — сохраняет изменения в истории
git log — показывает историю коммитов
git diff — показывает различия в файлах
git branch — показывает список веток
git switch — переключает ветки
git clone — копирует удалённый репозиторий
git push — отправляет коммиты на GitHub
git pull — получает изменения из удалённого репозитория
git remote -v — показывает связанные удалённые репозитории
